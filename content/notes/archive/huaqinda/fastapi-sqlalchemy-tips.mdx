## 可选筛选条件

重点在于建立一个可选的dict类型的传参filter

实际filter范例

```python
filter = {
	"category": "quick-note"
}
```

同时将其传入filter_by

```python
query.filter_by(**filter)
```

> [!TIP]
> filter_by必须在offset和limit之前

### 传参配置

```python
class SearchQuery(BaseModel):
    keyword: str
    page_num: int = 1
    page_size: int = 10
    filter: dict | None = None
```

### crud函数

```python

def search_document(db: Session, keyword: str, page_num: int, page_size: int, filter: dict | None = None):
    # 构建基础查询并先应用 filter
    query = db.query(models.document.Document) \
        .filter(models.document.Document.title.like(f'%{keyword}%'),
                models.document.Document.delete_at == None)

    # 如果 filter 参数不为空，动态添加额外的过滤条件
    if filter:
        query = query.filter_by(**filter)

    # 然后应用排序、分页和预加载
    query = query.order_by(models.document.Document.create_time.desc()) \
                 .offset((page_num - 1) * page_size) \
                 .limit(page_size) \
                 .options(selectinload(models.document.Document.users)) \
                 .options(selectinload(models.document.Document.labels))

    # 执行查询并返回结果
    db_documents = query.all()
    return db_documents
```

### router接口

```python
@document_router.post('/search', response_model=schemas.pagination.Pagination[schemas.document.DocumentInfo])
async def search_document(search_query: schemas.document.SearchQuery, db: Session = Depends(get_db)):
    count = crud.document.count_document(db=db, keyword=search_query.keyword, filter=search_query.filter)
    elements = crud.document.search_document(db=db,
                                             keyword=search_query.keyword,
                                             page_num=search_query.page_num,
                                             page_size=search_query.page_size,
                                             filter=search_query.filter)
    res = schemas.pagination.Pagination(
        total_elements=count,
        total_pages=math.ceil(count / search_query.page_size),
        page_num=search_query.page_num,
        page_size=search_query.page_size,
        current_page_elements=len(elements),
        elements=elements
    )
    return res
```
